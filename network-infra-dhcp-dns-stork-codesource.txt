bash

# CONFIG BIND9 – named.conf.local
/etc/bind/named.conf.local

text
zone "projet.local" {
  type master;
  file "/etc/bind/db.projet.local";
};

zone "1.168.192.in-addr.arpa" {
  type master;
  file "/etc/bind/db.192";
};
# ZONE DIRECTE
/etc/bind/db.projet.local

text
$TTL 604800
@ IN SOA ns1.projet.local. admin.projet.local. (
  2       ; Serial
  604800  ; Refresh
  86400   ; Retry
  2419200 ; Expire
  604800 ) ; Negative Cache TTL
@   IN NS ns1.projet.local.
ns1 IN A 192.168.1.10
pc1 IN A 192.168.1.100
pc2 IN A 192.168.1.101
# ZONE INVERSE
/etc/bind/db.192

text
$TTL 604800
@ IN SOA ns1.projet.local. admin.projet.local. (
  2       ; Serial
  604800  ; Refresh
  86400   ; Retry
  2419200 ; Expire
  604800 ) ; Negative Cache TTL
@    IN NS ns1.projet.local.
10   IN PTR ns1.projet.local.
100  IN PTR pc1.projet.local.
101  IN PTR pc2.projet.local.
# VÉRIFICATION BIND9

bash
named-checkconf
named-checkzone projet.local /etc/bind/db.projet.local
named-checkzone 1.168.192.in-addr.arpa /etc/bind/db.192
systemctl restart bind9
